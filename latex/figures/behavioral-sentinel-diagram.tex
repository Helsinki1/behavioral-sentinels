% Behavioral sentinel: the probe is carried inside the trajectory it measures.
\begin{tikzpicture}[
    font=\small,
    box/.style={rounded corners=2pt, draw, minimum height=8mm, inner xsep=6pt, align=center},
    turn/.style={box, fill=softgray, draw=agentblue!45, minimum width=22mm},
    sent/.style={box, fill=sentinelorange!12, draw=sentinelorange, minimum width=22mm},
    lbl/.style={font=\scriptsize\itshape, text=black!60},
    flow/.style={-{Stealth[length=2mm]}, draw=agentblue!70, thick},
    warn/.style={-{Stealth[length=2mm]}, draw=warningred, thick, dashed}
  ]
  \node[turn] (t1) {Turn $t$};
  \node[turn, right=7mm of t1] (t2) {Turn $t{+}1$};
  \node[turn, right=7mm of t2] (t3) {Turn $t{+}2$};
  \node[lbl, left=3mm of t1] {task};

  \node[sent, below=6mm of t1] (s1) {sentinel \checkmark};
  \node[sent, below=6mm of t2] (s2) {sentinel \checkmark};
  \node[sent, below=6mm of t3, fill=warningred!10, draw=warningred] (s3) {sentinel $\times$};
  \node[lbl, left=3mm of s1] {probe};

  \foreach \a/\b in {t1/t2, t2/t3} \draw[flow] (\a) -- (\b);
  \foreach \a/\b in {s1/s2, s2/s3} \draw[flow, draw=sentinelorange!70] (\a) -- (\b);
  \foreach \i in {1,2,3} \draw[draw=black!25, dotted] (t\i) -- (s\i);

  \node[box, right=9mm of t3, fill=recoverygreen!12, draw=recoverygreen,
        minimum width=24mm] (rec) {intervention\\+ recovery};
  \draw[warn] (s3.east) -- ++(6mm,0) |- (rec.south west);
  \node[lbl, below=1mm of rec.south, anchor=north] {alarm fires};

  \begin{scope}[on background layer]
    \node[fit=(t1)(s3), draw=black!20, rounded corners=3pt, inner sep=4mm] (traj) {};
  \end{scope}
  \node[lbl, above=0.5mm of traj.north east, anchor=south east]
       {the probe is \emph{inside} the measured trajectory};
\end{tikzpicture}
